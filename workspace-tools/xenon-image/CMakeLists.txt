cmake_minimum_required(VERSION 3.20)
project(TCSImageExport LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${XENONRECOMP_SOURCE}/XenonUtils")
    message(FATAL_ERROR "Set XENONRECOMP_SOURCE to the pinned XenonRecomp checkout")
endif()
add_subdirectory("${XENONRECOMP_SOURCE}" xenon-core EXCLUDE_FROM_ALL)
add_executable(tcs_image_export image_export.cpp)
target_compile_features(tcs_image_export PRIVATE cxx_std_17)
target_link_libraries(tcs_image_export PRIVATE XenonUtils)
set_property(TARGET tcs_image_export PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
